\newwrite\layout
\newcommand{\emitlayout}{%
  \convertcolorspec{named}{ink}{HTML}\inkhex
  \convertcolorspec{named}{muted}{HTML}\mutedhex
  \convertcolorspec{named}{rulegray}{HTML}\rulehex
  \immediate\openout\layout=review-table.json
  \immediate\write\layout{\openbrace}
  \immediate\write\layout{ "pageWidth": \pdfpoints{\paperwidth}, "pageHeight": \pdfpoints{\paperheight},}
  \immediate\write\layout{ "inset": \pdfpoints{\inset}, "top": \pdfpoints{\topinset}, "bottom": \pdfpoints{\bottominset},}
  \immediate\write\layout{ "columns": [\pdfpoints{\subjectwidth}, \pdfpoints{\answerwidth}, \pdfpoints{\sourcewidth}],}
  \immediate\write\layout{ "gutter": \pdfpoints{\gutter}, "rowSpace": \pdfpoints{\rowspace},}
  \immediate\write\layout{ "heavyRule": \pdfpoints{\heavyrulewidth}, "lightRule": \pdfpoints{\lightrulewidth},}
  \immediate\write\layout{ "ink": "\inkhex", "muted": "\mutedhex", "rule": "\rulehex",}
  \immediate\write\layout{ "heading": \headingSize, "body": \bodySize, "detail": \detailSize, "label": \labelSize}
  \immediate\write\layout{\closebrace}
  \immediate\closeout\layout}
